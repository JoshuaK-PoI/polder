-- Entrypoint: Program
Program := ( Statement )+

( Statement ) := 
		| ( VariableDeclarationStatement )
		| ( VariableAssignmentStatement )
		| ( IntrinsicStatement )
		| ( IfStatement )
		| ( WhileStatement )
		| ( ForeachStatement )
		| ( Comment )
		;

( Comment ) := 'P.S.' (.*)*\n ;

( VariableDeclarationStatement ) := LAAT ( Identifier ) VOORGOED? [ ( TypeDeclExpr ) | ( Expr ) ] ZIJN FULLSTOP;
( VariableAssignmentStatement ) := [
		| MAAK [dan] VAN ( Identifier ) ( Expr )
		| TEL [dan] ( Expr ) BIJ ( Identifier ) OP
		| TREK [dan] ( Expr ) VAN ( Identifier ) AF
		| VERMENIGVULDIG ( Identifier ) MET ( Expr )
		| DEEL ( Identifier ) DOOR ( Expr )
		| VERHOOG ( Identifier )
		| VERLAAG ( Identifier )
		| ( MethodCallStatement )
	] FULLSTOP;

( IntrinsicStatement ) := ( IntrinsicStart ) ( Expr ) (IntrinsicEnd(IntrinsicStart) ) FULLSTOP;
( IntrinsicStart ) := DRUK | SCHRIJF | LEES ;
( IntrinsicEnd(start) ) :=
		| start == DRUK     ? AF :
			start == SCHRIJF 	? OP :
			start == LEES 		? IN
		;

( IfStatement ) := ALS ( ComparisonExpr ) [ 
		| COMMA ( Statement -> Parser::InThenStatement = true ) ( ElseStatement )?
		| COLON ( Statement )+ ( ElseBlock )? KLAAR IS KEES FULLSTOP
	];

( ElseStatement ) := ZO NIET COMMA ( Statement -> Parser::InThenStatement = true );
( ElseBlock ) := ANDERS COLON ( Statement )+ ;

( WhileStatement ) := ZOLANG ( ComparisonExpr ) COLON ( Statement )+ KLAAR IS KEES FULLSTOP;
( ForeachStatement ) := VOOR ELKE [( Identifier ) EN]? ( Identifier ) IN ( Identifier ) COLON ( Statement )+ KLAAR IS KEES FULLSTOP;

( MethodCallStatement ) :=
	VOEG ( Expr ) AAN ( Identifier ) TOE;

( Expr ) :=
		| ( ArithExpr )
		| ( MethodCallExpr )
		| ( LiteralExpr )
		| ( Identifier )
		;

( ComparisonExpr ) := ( Expr ) ( Comparison ) ( Expr ) | ( Expr ) ( Literal ) IS;

( ArithExpr ) := ( Article ) ( ArithOp ) VAN ( Expr ) EN ( Expr );
( MethodCallExpr ) := ( Article ) ( Method ) VAN ( Identifier );
( TypeDeclExpr ) := ( Article ) ( Type ) ( LiteralExpr )?;

( LiteralExpr ) :=
		| ( StringLiteral )
		| ( IntegerLiteral )
		| ( IntegerWord )
		| ( FloatLiteral )
		| ( CharLiteral )
		| TRUE
		| FALSE
		;

( Comparison ) := NIET? [ [ GROTER | KLEINER ] IS DAN | GELIJK IS AAN ];

( StringLiteral ) := ( OpenQuote )[([^( CloseQuote )])*]( CloseQuote );
( IntegerLiteral ) := [0-9]+;
( IntegerWord ) = [a-z].*; /* IntegerWord is codified as the full text representation in Dutch of a number. In the tokenization it is marked as an IntegerLiteral. */
( FloatLiteral ) := [0-9]+ COMMA [0-9]+;
( CharLiteral ) := QUOTE [([^QUOTE])] QUOTE; /* Any valid UTF8-character */

( VarOrNumber ) :=
	| ( Identifier )
	| ( IntegerLiteral )
	| ( FloatLiteral )
	;

( Identifier ) := [a-zA-Z][a-zA-Z0-9_]*;

( Article ) := 'de' | 'het' | 'een';
( TypeName ) := 'getal' | 'kommagetal' | 'tekst' | 'waarheid' | 'letter' | ['lijst' VAN ( TypePlural )?];
( TypePlural ) := 'getallen' | 'kommagetallen' | 'teksten' | 'waarheden' | 'letters' | ['lijsten' VAN ( TypePlural )];

( ArithOp ) := SOM | VERSCHIL | PRODUCT | QUOTIENT | MODULO;
( OpenQuote ) := '"' | '“' | '„';
( CloseQuote ) := '"' | '”';

[dan] := Parser::InThenStatement ? DAN : '' ;
